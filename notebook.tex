\subsection{\texorpdfstring{השדות
$B,H,M$
ופרמאביליות}{השדות
B,H,M
ופרמאביליות}}
בחומר שבו
$\vec{M}=\chi_{\mathrm{M}}\vec{H}$
(
$\chi_{\mathrm{M}}$
הסוספטיביליות המגנטית),
\begin{equation}
\vec{B}=\mu_{0}(\vec{H}+\vec{M})=\mu_{0}(1+\chi_{\mathrm{M}})\vec{H}\equiv\mu\vec{H},
\end{equation}
ואנו עובדים ביחידות
\textenglish{MKS}:
$B$
בטסלה,
$H$
ב-
\si{\ampere\per\meter}
ו-
$\mu$
ב-
\si{\tesla\meter\per\ampere}.
מכאן
\begin{equation}
\mu(H)\equiv\frac{B}{H},\qquad
\mu_{\mathrm{r}}(H)\equiv\frac{\mu(H)}{\mu_{0}}=\frac{B}{\mu_{0}H}.
\end{equation}
בחומר לינארי
$\mu_{\mathrm{r}}$
קבוע; בפרומגנט
$\mu_{\mathrm{r}}=\mu_{\mathrm{r}}(H)$
ויש לולאת חשל.

\subsection{חילוץ פרמאביליות הריק}
עבור טבעת ברזל באורך
$L$
עם רווח בעובי
$L'$
ממולא נחושת (
$\mu_{\mathrm{Cu}}\approx\mu_{0}$
), חוק אמפר נותן
\begin{equation}
\oint\vec{H}\cdot\mathrm{d}\vec{\ell}=H_{\mathrm{Fe}}\,L+H_{\mathrm{gap}}\,L'=NI.
\end{equation}
בהנחת רציפות
$B_{\mathrm{Fe}}=B_{\mathrm{gap}}\equiv B$
(רלוונטית ברוויה),
\begin{equation}\label{eq:linear}
\frac{NI}{B}=\frac{1}{\mu_{0}}\,L'+\frac{L}{\mu_{\mathrm{Fe}}(H)}.
\end{equation}
אם
$B$
נשמר קבוע בסריקת
$L'$
, ובתחום רוויה
$\mu_{\mathrm{Fe}}(H)$
בקירוב קבוע, השיפוע הוא
$1/\mu_0$.

\subsection{\texorpdfstring{גזירת
$H(V_{x})$
ו-
$B(V_{y})$
}{גזירת
H(Vx)
ו-
B(Vy)}}
חוק אמפר
$HL=N_1 I_x$
ביחד עם
$I_x=V_x/R_x$
נותן
\begin{equation}\label{eq:Hcal}
H=\frac{N_{1}}{L\,R_{x}}\,V_{x}.
\end{equation}
חוק פאראדיי
$V_{\mathrm{ind}}=-N_2 A\,\frac{dB}{dt}$
ובחירת
$\omega R_yC\gg1$
הופכים את הקבל לאינטגרטור,
\begin{equation}\label{eq:Bcal}
V_{y}\approx\frac{1}{R_{y}C}\int V_{\mathrm{ind}}\,\mathrm{d}t=\frac{N_{2}A}{R_{y}C}\,B\;\Longrightarrow\;B=\frac{R_{y}C}{N_{2}A}\,V_{y}.
\end{equation}
בפועל
$\tau=R_yC=\SI{0.223}{\second}$
ו-
$f=\SI{50}{\hertz}$
, כך ש-
$\omega\tau\approx70$.

הקריאות
$\Delta V_x,\Delta V_y$
נמדדות שיא־לשיא, ולכן בחזרה על~\eqref{eq:Hcal}--\eqref{eq:Bcal} עם
$\Delta V$
נכנס גורם
$1/2$:
\begin{equation}\label{eq:HBpp}
H=\frac{N_{1}}{2\,L\,R_{x}}\,\Delta V_{x},\qquad B=\frac{R_{y}C}{2\,N_{2}A}\,\Delta V_{y}.
\end{equation}

הגדלים הנגזרים
$L=2(a_L+b_L)$
ו-
$A=a_Ab_A$
מורכבים משתי קריאות סרגל בלתי תלויות, ולכן
\begin{equation}
\sigma_L=2\sqrt{\sigma_{a_L}^2+\sigma_{b_L}^2},\qquad
\sigma_A=A\sqrt{\left(\frac{\sigma_{a_A}}{a_A}\right)^2+
\left(\frac{\sigma_{b_A}}{b_A}\right)^2}.
\end{equation}
אי־ודאויות
$H$,
$B$,
$\mu_{\mathrm{r}}$
מתקבלות משגיאה עקיפה לפי משוואה~\eqref{eq:HBpp} ו-
$\mu_{\mathrm{r}}=B/(\mu_0 H)$:
\begin{align}
\sigma_H&=H\sqrt{\left(\frac{\sigma_{\Delta V_x}}{\Delta V_x}\right)^2+
\left(\frac{\sigma_L}{L}\right)^2+
\left(\frac{\sigma_{R_x}}{R_x}\right)^2},\\
\sigma_B&=B\sqrt{\left(\frac{\sigma_{\Delta V_y}}{\Delta V_y}\right)^2+
\left(\frac{\sigma_{R_y}}{R_y}\right)^2+
\left(\frac{\sigma_C}{C}\right)^2+
\left(\frac{\sigma_A}{A}\right)^2},\\
\sigma_{\mu_{\mathrm{r}}}&=\mu_{\mathrm{r}}
\sqrt{\left(\frac{\sigma_B}{B}\right)^2+
\left(\frac{\sigma_H}{H}\right)^2}.
\end{align}

\section{פרמאביליות הריק}

\subsection{מהלך הניסוי}
בשתי סדרות
$V_y$
(
$\propto B$
) הוחזק קבוע ולוחיות הנחושת הוסרו בזו אחר זו, כדי לשמור על
$B$
קבוע בסריקת
$L'$.
ערך
$B$
נבחר קרוב לרוויה תחת מגבלות הזרם. עוביי הרווח:
$L'=\SIlist{0.35;0.20;0.10;0}{\milli\meter}$
בסדרה
\texttt{a},
$\SIlist{0.50;0.45;0.35;0.20;0.10;0}{\milli\meter}$
בסדרה
\texttt{b}.
בכל סדרה נרשם אותו
$\Delta V_y$
בכל נקודות הרווח, ולכן
$B$
קבוע בקירוב: בסדרה
\texttt{a}
$\Delta V_y=\SI{3.67}{\volt}$
ו-
$B=\SI{1.02}{\tesla}$
ובסדרה
\texttt{b}
$\Delta V_y=\SI{2.83}{\volt}$
ו-
$B=\SI{0.79}{\tesla}$
בהתאמה. ערכי
$B$
אלה נמוכים מערך השיא של חלק א' (
$\SI{1.24}{\tesla}$
). לפי איור~\ref{fig:BH} זהו תחום שבו העקומה כבר מתמתנת, אך לא מובטח שהברזל ברוויה מלאה; לכן אי־רוויה אפשרית היא מקור שיטתי מרכזי.

$T_C$.

\subsection{הכנת הנתונים}
בעיבוד המעודכן לא בוצע חיתוך התייצבות שדה נוסף: כל הלולאות שנרשמו בכל סדרה נכנסו לחילוץ שלושת מדדי המגנוט. שינויי אמפליטודת השדה בתחילת הסדרה נשארים לכן חלק מן הנתונים, ולא מוגדר סף טכני שמסיר לולאות.

כדי שמדד שיטה~2 עדיין ייצג את קריאת קצה הלולאה של
\textenglish{LabVIEW}
בלי לכפות חלון שדה משותף לכל הטמפרטורות, בכל לולאה קראנו אותו ב-
\begin{equation}
V_{x,\mathrm{sat}}(T)=0.98\,V_{x,\max}(T),
\end{equation}
ובענף השלילי ב-
$-V_{x,\mathrm{sat}}(T)$.
הגורם
$0.98$
מזיז את הקריאה מעט פנימה מקצה הסריקה כדי שהאינטרפולציה המקומית תהיה יציבה, אך אינו משמש לסינון לולאות.

תחומי הטמפרטורה ששימשו בעיבוד הם
\textenglish{A}:
$\SI{82.9}{\kelvin}$--$\SI{273.3}{\kelvin}$,
\textenglish{B}:
$\SI{83.1}{\kelvin}$--$\SI{273.2}{\kelvin}$,
\textenglish{C}:
$\SI{110.7}{\kelvin}$--$\SI{273.3}{\kelvin}$.
סדרה~\textenglish{C}
עדיין נשמרת כבדיקת רגישות, משום שהשדה בה נמוך בהרבה והתחום הקר שלה קצר מזה של סדרות~\textenglish{A,B}.

\subsection{חילוץ \texorpdfstring{
$M(T)$
}{M(T)} מן הלולאות}
לפני החילוץ הוסר רקע לינארי שנקבע מן הרבעון העליון של טמפרטורות הסדרה (רקע אמפירי, לא הנחת תחום קירי--וייס נקי).
$M$
להלן הוא מדד מגנטיזציה יחסי: חצי פתיחת הלולאה בציר
$V_y$
לאחר חיסור הרקע.

השתמשנו בשלושה מדדים מאותה לולאה:
\begin{itemize}
\item
$M_0$:
פתיחת הלולאה ב-
$V_x=0$.
\item
$M_{\mathrm{sat}}$:
פתיחת הלולאה בקצה השדה של אותה לולאה,
$V_{x,\mathrm{sat}}(T)=0.98V_{x,\max}(T)$.
כמקובל בדוחות עבר נקרא לגודל זה מדד רוויה, אך בפועל זו אינה בהכרח רוויה מלאה אלא קירוב הנקבע על ידי תחום השדה שנמדד.
\item
$M_0^{\mathrm{ext}}$:
אקסטרפולציה של זנבות הרוויה אל
$V_x=0$.
\end{itemize}

בכל טמפרטורה ללולאה שני ענפים. הסימונים הבאים אינם נקודת ענף יחידה אלא חצי הפרש סימטרי בין הענפים, מה שמבטל היסט משותף בציר האנכי:
\begin{align}
M_{0}(T)&=\frac{1}{2}\left|M_{+}(T;0)-M_{-}(T;0)\right|,\\
M_{\mathrm{sat}}(T)&=\frac{1}{2}\left|M_{+}(T;V_{x,\mathrm{sat}}(T))-M_{-}(T;-V_{x,\mathrm{sat}}(T))\right|,\\
M_{0}^{\mathrm{ext}}(T)&=\frac{1}{2}\left|b^{(+)}-b^{(-)}\right|.
\end{align}
בשיטה~3,
$b^{(\pm)}$
הם חיתוכי התאמות לינאריות קצרות בזנבות הרוויה,
$M_{\pm}(V_x)\approx b^{(\pm)}(T)+a^{(\pm)}(T)V_x$.
זהו ניסוח מעשי של התאמת זנב לינארי והמשך הקו ל-
$V_x=0$.
חלון ההתאמה נבחר מזנבות הלולאה בלבד: מתחילים מחמש נקודות הקצה, שומרים נקודות עם
$|M|\ge0.95|\hat M_0^{\mathrm{ext}}|$
, ומרחיבים פנימה כל עוד השיפוע נשאר יציב.

כל שלושת הגדלים נורמלו באותה צורה:
\begin{equation}
y_{\mathrm{norm}}(T)=\frac{y(T)-\min_T y}{\max_T y-\min_T y}.
\end{equation}

הדגמה של השיטות על לולאה אחת באיור~\ref{fig:curie-method3-tail-example}; המדדים המנורמלים באיור~\ref{fig:curie-method123}.

\subsection{אי־ודאויות}

$V_x,V_y,T$;

$\tilde M=1/2$.

$T_C$.

למתחי הסקופ השתמשנו באי־ודאות רזולוציה בלבד:
\begin{equation}
\sigma_{V_x}=\frac{q_x}{\sqrt{12}},
\qquad
\sigma_{V_y}=\frac{q_y}{\sqrt{12}}.
\end{equation}
לטמפרטורה של לולאה אחת צירפנו רזולוציית דיווח, דיוק תרמומטר, וטשטוש בגלל שינוי הטמפרטורה בזמן רכישת הלולאה:
\begin{align}
\sigma_T&=\sqrt{\left(\frac{q_T}{\sqrt{12}}\right)^2+
\sigma_{\mathrm{therm}}^2+\sigma_{\mathrm{smear}}^2},\qquad
\sigma_{\mathrm{smear}}=\frac{|\dot T|\Delta t_{\mathrm{loop}}}{\sqrt{12}},\\
\sigma_{\mathrm{therm}}(T)&=0.001\,|T_{^\circ\mathrm{C}}|+
\begin{cases}
\SI{1}{\kelvin}, & T_{^\circ\mathrm{C}}\ge\SI{-60}{\degreeCelsius},\\
\SI{2}{\kelvin}, & T_{^\circ\mathrm{C}}<\SI{-60}{\degreeCelsius}.
\end{cases}
\end{align}
בנתונים
$\Delta t_{\mathrm{loop}}\simeq\SI{5.6}{\second}$,
וסביב המעבר
$\sigma_{\mathrm{therm}}\simeq\SI{1.06}{\kelvin}$.

מדדי המגנוט חושבו מחצי הפרש בין שני ענפים, ולכן אי־הוודאות שלהם היא חצי סכום ריבועי של אי־ודאויות הענפים. לשיטות הישירות
($M_0,M_{\mathrm{sat}}$)
השתמשנו ב-
\begin{equation}
\sigma_M=\frac{1}{2}\sqrt{\sigma_{+}^2+\sigma_{-}^2},
\end{equation}
כאשר
$\sigma_{+},\sigma_{-}$
התקבלו מהתאמות
\textenglish{ODR}
מקומיות על שני הענפים. לשיטת הזנבות השתמשנו באי־ודאות חיתוכי הזנבות,
\begin{equation}
\sigma_{M_0^{\mathrm{ext}}}=\frac{1}{2}\sqrt{\sigma_{b^{(+)}}^2+\sigma_{b^{(-)}}^2}.
\end{equation}
אחרי הנרמול
$\tilde M=(M-M_{\min})/(M_{\max}-M_{\min})$
קיבלנו
\begin{equation}
\sigma_{\tilde M}=\frac{\sigma_M}{M_{\max}-M_{\min}},
\end{equation}
כאשר
$M_{\min},M_{\max}$
נלקחו כקבועי סדרת העיבוד.

הטמפרטורה המקומית
$T_C$
נקראה מאינטרפולציה לינארית בין שתי הנקודות המקיפות את
$\tilde M=1/2$:
\begin{equation}
T_C=T_i+\frac{1/2-\tilde M_i}{\tilde M_{i+1}-\tilde M_i}(T_{i+1}-T_i).
\end{equation}
אי־הוודאות המקומית של חצייה זו חושבה מהנגזרות לפי שתי הטמפרטורות ושני ערכי המגנוט המנורמל:
\begin{equation}
\sigma_{T_C,\mathrm{loc}}^2=
\sum_{j=i,i+1}\left(\frac{\partial T_C}{\partial T_j}\sigma_{T_j}\right)^2+
\sum_{j=i,i+1}\left(\frac{\partial T_C}{\partial \tilde M_j}\sigma_{\tilde M_j}\right)^2.
\label{eq:curie-sigmaTc-local}
\end{equation}
האי־ודאות הסופית של התוצאה המדווחת כוללת מעבר לכך גם את רגישות החילוץ לשיטה ולסדרת מדידה:
\begin{equation}
\sigma_{T_C,\mathrm{final}}=\sqrt{\sigma_{T_C,\mathrm{loc}}^2+
\sigma_{\mathrm{method}}^2+
\sigma_{\mathrm{series}}^2}.
\end{equation}
לכן האי־ודאויות בסוגריים בטבלאות התוצאות הן אי־ודאויות חצייה מקומיות בלבד, ואילו הערך הסופי מעוגל כך שיכלול גם את פיזור השיטות והסדרות.

\section{דיון ומסקנות}
הרכב סגסוגת המונל אינו ידוע, ולכן איכות התוצאה נבחנת בעיקר מעקביות פנימית. האומדן המרכזי נלקח מהממוצע המשוקלל של שלוש חציות חצי־הגובה בסדרה~
\textenglish{A};
האי־ודאות נשלטת על ידי הפיזור בין השיטות, ובעיקר על ידי רגישות שיטה~3 לבחירת זנבות הרוויה:
\begin{equation}\label{eq:Tc-headline}
T_C=\SI{198(18)}{\kelvin}=\SI{-75(18)}{\degreeCelsius},
\end{equation}
כאשר האי־ודאות כוללת את פיזור השיטות, ההבדל בין סדרות
\textenglish{A,B}
ואי־ודאות טמפרטורה מסדר
$\SI{1}{\kelvin}$--$\SI{2}{\kelvin}$.
בקירוב, החציה המקומית תורמת כ-
$\SI{1}{\kelvin}$,
פיזור השיטות בסדרה~\textenglish{A}
כ-
$\SI{18}{\kelvin}$,
ופיזור סדרות~\textenglish{A,B}
כ-
$\SI{1}{\kelvin}$.
זהו אומדן ניסויי לטמפרטורת קירי של הדגימה בשדה סופי, כלומר הטמפרטורה שחולצה מחצי־הגובה של דעיכת המגנוט ולא ערך תרמודינמי אידיאלי בשדה אפס.

\begin{itemize}
\item התצפית העיקרית: הלולאות נסגרות עם החימום, והתגובה נעשית כמעט לינארית מעל המעבר.
\item שתי השיטות הישירות בסדרה~
\textenglish{A}
ובסדרה~\textenglish{B}
נותנות תחום מעבר סביב
$\SI{195}{\kelvin}$--$\SI{204}{\kelvin}$;
שיטה~3 נמוכה יותר ולכן משמשת בדיקת רגישות לחלון זנבות הרוויה.
\item שינוי הזרם הראשוני משנה את פתיחת הלולאה ומזיז את אומדן
$T_C$
בכמה עד עשרות קלווין; סדרה~
\textenglish{C}
נמדדה בשדה נמוך בהרבה ולכן אינה משמשת לאומדן המרכזי.
\item מקורות הטיה עיקריים: השהיה תרמית וגרדיאנטים תרמיים, חיסור רקע לינארי, אמפליטודת השדה הסופית, ובחירת חלון זנב.
\item במדידה חוזרת: למדוד את גאומטריית הדגימה, לשפר הצמדה תרמית, לסרוק לאט יותר, לכסות תחום קר יציב בכל סדרה, ולכייל את ערוץ
$B$
עצמאית.
\end{itemize}
