\section{רקע תיאורטי}

\subsection{\texorpdfstring{השדות
$B,H,M$
ופרמאביליות}{השדות
B,H,M
ופרמאביליות}}
בחומר שבו
$\vec{M}=\chi_{\mathrm{M}}\vec{H}$
(
$\chi_{\mathrm{M}}$
הסוספטיביליות המגנטית),
\begin{equation}
\vec{B}=\mu_{0}(\vec{H}+\vec{M})=\mu_{0}(1+\chi_{\mathrm{M}})\vec{H}\equiv\mu\vec{H},
\end{equation}
ואנו עובדים ביחידות
\textenglish{MKS}:
$B$
בטסלה,
$H$
ב-
\si{\ampere\per\meter}
ו-
$\mu$
ב-
\si{\tesla\meter\per\ampere}.
מכאן
\begin{equation}
\mu(H)\equiv\frac{B}{H},\qquad
\mu_{\mathrm{r}}(H)\equiv\frac{\mu(H)}{\mu_{0}}=\frac{B}{\mu_{0}H}.
\end{equation}
בחומר לינארי
$\mu_{\mathrm{r}}$
קבוע; בפרומגנט
$\mu_{\mathrm{r}}=\mu_{\mathrm{r}}(H)$
ויש לולאת חשל.

$H$

\subsection{חילוץ פרמאביליות הריק}
עבור טבעת ברזל באורך מגנטי
$L$
עם רווח בעובי
$L'$
ממולא נחושת (
$\mu_{\mathrm{Cu}}\approx\mu_{0}$
), חוק אמפר נותן
\begin{equation}
\oint\vec{H}\cdot\mathrm{d}\vec{\ell}=H_{\mathrm{Fe}}\,L+H_{\mathrm{gap}}\,L'=NI.
\end{equation}
בהנחת רציפות
$B_{\mathrm{Fe}}=B_{\mathrm{gap}}\equiv B$
(רלוונטית ברוויה),
\begin{equation}\label{eq:linear}
\frac{NI}{B}=\frac{1}{\mu_{0}}\,L'+\frac{L}{\mu_{\mathrm{Fe}}(H)}.
\end{equation}
אם
$B$
נשמר קבוע בסריקת
$L'$
, ובתחום רוויה
$\mu_{\mathrm{Fe}}(H)$
בקירוב קבוע, השיפוע הוא
$1/\mu_0$.

$V_x\propto I_x\propto H$.

$V_x,V_y$
מוזנים לצירי
$X,Y$

\subsection{\texorpdfstring{גזירת
$H(V_{x})$
ו-
$B(V_{y})$
}{גזירת
H(Vx)
ו-
B(Vy)}}
חוק אמפר
$HL=N_1 I_x$
ביחד עם
$I_x=V_x/R_x$
נותן
\begin{equation}\label{eq:Hcal}
H=\frac{N_{1}}{L\,R_{x}}\,V_{x}.
\end{equation}
חוק פאראדיי
$V_{\mathrm{ind}}=-N_2 A\,dB/dt$
ובחירת
$\omega R_yC\gg1$
הופכים את הקבל לאינטגרטור,
\begin{equation}\label{eq:Bcal}
V_{y}\approx\frac{1}{R_{y}C}\int V_{\mathrm{ind}}\,\mathrm{d}t=\frac{N_{2}A}{R_{y}C}\,B\;\Longrightarrow\;B=\frac{R_{y}C}{N_{2}A}\,V_{y}.
\end{equation}
בפועל
$\tau=R_yC=\SI{0.223}{\second}$
ו-
$f=\SI{50}{\hertz}$
, כך ש-
$\omega\tau\approx70$.

הקריאות
$\Delta V_x,\Delta V_y$
נמדדות שיא־לשיא, ולכן בחזרה על~\eqref{eq:Hcal}--\eqref{eq:Bcal} עם
$\Delta V$
נכנס גורם
$1/2$:
\begin{equation}\label{eq:HBpp}
H=\frac{N_{1}}{2\,L\,R_{x}}\,\Delta V_{x},\qquad B=\frac{R_{y}C}{2\,N_{2}A}\,\Delta V_{y}.
\end{equation}

הגדלים הנגזרים
$L=2(a_L+b_L)$
ו-
$A=a_Ab_A$
מורכבים משתי קריאות סרגל בלתי תלויות, ולכן
\begin{equation}
\sigma_L=2\sqrt{\sigma_{a_L}^2+\sigma_{b_L}^2},\qquad
\sigma_A=A\sqrt{\left(\frac{\sigma_{a_A}}{a_A}\right)^2+
\left(\frac{\sigma_{b_A}}{b_A}\right)^2}.
\end{equation}
אי־ודאויות
$H$,
$B$,
$\mu_{\mathrm{r}}$
מתקבלות משגיאה עקיפה לפי משוואה~\eqref{eq:HBpp} ו-
$\mu_{\mathrm{r}}=B/(\mu_0 H)$:
\begin{align}
\sigma_H&=H\sqrt{\left(\frac{\sigma_{\Delta V_x}}{\Delta V_x}\right)^2+
\left(\frac{\sigma_L}{L}\right)^2+
\left(\frac{\sigma_{R_x}}{R_x}\right)^2},\\
\sigma_B&=B\sqrt{\left(\frac{\sigma_{\Delta V_y}}{\Delta V_y}\right)^2+
\left(\frac{\sigma_{R_y}}{R_y}\right)^2+
\left(\frac{\sigma_C}{C}\right)^2+
\left(\frac{\sigma_A}{A}\right)^2},\\
\sigma_{\mu_{\mathrm{r}}}&=\mu_{\mathrm{r}}
\sqrt{\left(\frac{\sigma_B}{B}\right)^2+
\left(\frac{\sigma_H}{H}\right)^2}.
\end{align}

\section{פרמאביליות הריק}

\subsection{מהלך הניסוי}
בשתי סדרות
$V_y$
(
$\propto B$
) הוחזק קבוע ולוחיות הנחושת הוסרו בזו אחר זו, כדי לשמור על
$B$
קבוע בסריקת
$L'$.
ערך
$B$
נבחר קרוב לרוויה תחת מגבלות הזרם. עוביי הרווח:
$L'=\SIlist{0.35;0.20;0.10;0}{\milli\meter}$
בסדרה
\texttt{a},
$\SIlist{0.50;0.45;0.35;0.20;0.10;0}{\milli\meter}$
בסדרה
\texttt{b}.
בכל סדרה נרשם אותו
$\Delta V_y$
בכל נקודות הרווח, ולכן
$B$
קבוע בקירוב: בסדרה
\texttt{a}
$\Delta V_y=\SI{3.67}{\volt}$
ו-
$B=\SI{1.02}{\tesla}$
ובסדרה
\texttt{b}
$\Delta V_y=\SI{2.83}{\volt}$
ו-
$B=\SI{0.79}{\tesla}$
בהתאמה. ערכי
$B$
אלה נמוכים מערך השיא של חלק א' (
$\SI{1.24}{\tesla}$
). לפי איור~\ref{fig:BH} זהו תחום שבו העקומה כבר מתמתנת, אך לא מובטח שהברזל ברוויה מלאה; לכן אי־רוויה אפשרית היא מקור שיטתי מרכזי.

\subsection{עיבוד נתונים}
התאמנו את
$NI/B$
כפונקציה של
$L'$.
לפי~\eqref{eq:linear} השיפוע הוא
$1/\mu_0$
והחיתוך מייצג את תרומת הברזל האפקטיבית.
עבור כל נקודה חושבו
$I=\Delta V_x/(2R_x)$
ו-
$B=R_yC\Delta V_y/(2NA)$
וכן חושבה אי־הוודאות ב-
$NI/B$
מתוך התפשטות שגיאות של
$\Delta V_x,\Delta V_y,R_x,R_y,C,A$.

\subsection{מעבר פאזה}
מעל
$T_C$
תנודות תרמיות שוברות את הסידור הספונטני, והחומר עובר לפאזה פאראמגנטית (
$M=0$

$H=0$

$T_C$

$M$

\subsection{חוק קירי--וייס}
בפאזה הפאראמגנטית (
$T>T_C$

\begin{equation}
\chi_{\mathrm{M}}=\frac{M}{H}=\frac{C_{\mathrm{CW}}}{T-T_C},
\end{equation}
ו-
$1/\chi_{\mathrm{M}}(T)$
לינארי, חותך את ציר
$T$
ב-
$T_C$.
בפועל לא השתמשנו בהתאמת
$1/\chi_{\mathrm{M}}(T)$:
הגאומטריה של הסלילים והדגימה לא נמדדה מספיק טוב להמרה אבסולוטית ל-
$H,M$

\subsection{קירוב שדה ממוצע ליד המעבר}

\end{equation}

$M^2(T)$
לינארי סמוך ל-
$T_C$.
בפועל לא ביצענו התאמת
$M^2(T)$
משום שהמדידה אינה מכוילת אבסולוטית והשדה הסופי מרחיב את המעבר.

בשדה אפס פרמטר הסדר נעלם בחדות ב-
$T_C$
; בשדה סופי המגנטיזציה המושרת לא מתאפסת בדיוק במעבר והמעבר מתרחב (איור~\ref{fig:curie-finite-field-schematic}).

\subsection{חילוץ \texorpdfstring{
$M(T)$
}{M(T)} מן הלולאות}
לפני החילוץ הוסר רקע לינארי שנקבע מן הרבעון העליון של טמפרטורות הסדרה (רקע אמפירי, לא הנחת תחום קירי--וייס נקי).
$M$
להלן הוא מדד מגנטיזציה יחסי: חצי פתיחת הלולאה בציר
$V_y$
לאחר חיסור הרקע.

השתמשנו בשלושה מדדים מאותה לולאה:
\begin{itemize}
\item
$M_{\mathrm{r}}$:
פתיחת הלולאה ב-
$V_x=0$.
\item
$M_{\mathrm{sat}}$:
פתיחת הלולאה בקצה חלון השדה המשותף. כמקובל בדוחות עבר נקרא לגודל זה מדד רוויה, אך בפועל זו אינה בהכרח רוויה מלאה אלא קירוב הנקבע על ידי תחום השדה שנמדד.
\item
$M_0$:
אקסטרפולציה של זנבות הרוויה אל
$V_x=0$.
\end{itemize}

בכל טמפרטורה ללולאה שני ענפים. הסימונים הבאים אינם נקודת ענף יחידה אלא חצי הפרש סימטרי בין הענפים, מה שמבטל היסט משותף בציר האנכי:
\begin{align}
M_{\mathrm{r}}(T)&=\frac{1}{2}\left|M_{+}(T;0)-M_{-}(T;0)\right|,\\
M_{\mathrm{sat}}(T)&=\frac{1}{2}\left|M_{+}(T;V_{x,\mathrm{sat}})-M_{-}(T;-V_{x,\mathrm{sat}})\right|,\\
M_{0}(T)&=\frac{1}{2}\left|b^{(+)}-b^{(-)}\right|.
\end{align}
בשיטה~3,
$b^{(\pm)}$
הם חיתוכי התאמות לינאריות קצרות בזנבות הרוויה,
$M(V_x)\approx M_{0}(T)+a(T)V_x$.
זהו ניסוח מעשי של התאמת זנב לינארי והמשך הקו ל-
$V_x=0$.
חלון ההתאמה נבחר מזנבות הלולאה בלבד: מתחילים מחמש נקודות הקצה, שומרים נקודות עם
$|M|\ge0.95|\hat M_0|$
, ומרחיבים פנימה כל עוד השיפוע נשאר יציב.

כל שלושת הגדלים נורמלו באותה צורה:
\begin{equation}
y_{\mathrm{norm}}(T)=\frac{y(T)-\min_T y}{\max_T y-\min_T y}.
\end{equation}

\subsection{אי־ודאויות}
בנרמול לחצי־גובה, כל סקאלה משותפת של
$M$
(
$N,L,R_x,R_y,C,A$
) מבטלת את עצמה. אי־הוודאות המקומית של כל חציית חצי־גובה חושבה מהפיזור בהתאמות המקומיות ללולאה ומרזולוציות הדיגיטציה בצירים
$q_x=\SI{0.6}{\milli\volt}$,
$q_y=\SI{0.4}{\milli\volt}$.
בנוסף יש תרומה משותפת של מד הטמפרטורה והצמד התרמי, מסדר
$\SI{1}{\kelvin}$--$\SI{2}{\kelvin}$,
וכן טשטוש קטן בזמן התחממות הלולאה. לכן האי־ודאויות בסוגריים בטבלאות הן אי־ודאויות חילוץ מקומיות, ואילו האי־ודאות הסופית נקבעת גם מפיזור השיטות והסדרות.

\section{דיון ומסקנות}
הרכב סגסוגת המונל אינו ידוע, ולכן איכות התוצאה נבחנת בעיקר מעקביות פנימית. האומדן המרכזי נלקח מהממוצע המעוגל של שתי השיטות הישירות בסדרה~
\textenglish{A}
(
$M_{\mathrm{r}},M_{\mathrm{sat}}$
), שהסכימו בתוך
$\SI{2}{\kelvin}$:
\begin{equation}\label{eq:Tc-headline}
T_C=\SI{216(10)}{\kelvin}=\SI{-58(10)}{\degreeCelsius},
\end{equation}
כאשר האי־ודאות כוללת את פיזור השיטות, ההבדל בין סדרות
\textenglish{A,B}
ואי־ודאות טמפרטורה מסדר
$\SI{1}{\kelvin}$--$\SI{2}{\kelvin}$.
בקירוב, החציה המקומית תורמת כ-
$\SI{1}{\kelvin}$,
פיזור הסדרות הישירות כ-
$\SI{6}{\kelvin}$,
והעיגול ל-
$\SI{10}{\kelvin}$
שומר מקום להטיות שיטתיות ולצמד התרמי.
זהו אומדן ניסויי לטמפרטורת קירי של הדגימה בשדה סופי, כלומר הטמפרטורה שחולצה מחצי־הגובה של דעיכת המגנוט ולא ערך תרמודינמי אידיאלי בשדה אפס.

\begin{itemize}
\item התצפית העיקרית: הלולאות נסגרות עם החימום, והתגובה נעשית כמעט לינארית מעל המעבר.
\item שתי השיטות הישירות בסדרה~
\textenglish{A}
נותנות את אותו תחום מעבר; שיטה~3 נמוכה יותר ולכן משמשת בדיקת רגישות לחלון זנבות הרוויה.
\item שינוי הזרם הראשוני משנה את פתיחת הלולאה ומזיז את אומדן
$T_C$
בכמה קלווין; סדרה~
\textenglish{C}
חסרה תחום קר יציב ולכן אינה משמשת לאומדן המרכזי.
\item מקורות הטיה עיקריים: השהיה תרמית וגרדיאנטים תרמיים, חיסור רקע לינארי, חיתוך התייצבות, ובחירת חלון זנב.
\item במדידה חוזרת: למדוד את גאומטריית הדגימה, לשפר הצמדה תרמית, לסרוק לאט יותר, לכסות תחום קר יציב בכל סדרה, ולכייל את ערוץ
$B$
עצמאית.
\end{itemize}
